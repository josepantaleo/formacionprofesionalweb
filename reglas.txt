// presencia y aportes.

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================================================
    // FUNCIONES GENERALES
    // =========================================================

    function signedIn() {
      return request.auth != null;
    }

    function ownsDocument(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function isActiveStudent(uid) {
      return ownsDocument(uid) &&
        exists(/databases/$(database)/documents/estudiantes/$(uid)) &&
        get(
          /databases/$(database)/documents/estudiantes/$(uid)
        ).data.get('estadoCuenta', 'pendiente') == 'activo';
    }

    function verifiedEmail() {
      return signedIn() &&
        request.auth.token.get('email_verified', false) == true;
    }

    function authenticatedEmail() {
      return request.auth.token.get('email', '');
    }

    function isPrincipalTeacherAdmin() {
      return verifiedEmail() &&
        authenticatedEmail() == 'ipem146centenario@gmail.com';
    }

    function isManagedTeacher() {
      return verifiedEmail() &&
        exists(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ) &&
        get(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ).data.get('email', '') == authenticatedEmail() &&
        get(
          /databases/$(database)/documents/docentesAutorizados/$(authenticatedEmail())
        ).data.get('activo', false) == true;
    }

    function isTeacher() {
      return isPrincipalTeacherAdmin() || isManagedTeacher();
    }


    // =========================================================
    // DOCENTES
    // =========================================================

    function validTeacherRecord(docenteId) {
      return request.resource.data.keys().hasAll([
          'email',
          'nombre',
          'rol',
          'activo',
          'creadoEn',
          'creadoPor',
          'actualizadoEn',
          'actualizadoPor',
          'bajaEn',
          'bajaPor',
          'motivoBaja'
        ]) &&
        request.resource.data.keys().hasOnly([
          'email',
          'nombre',
          'rol',
          'activo',
          'creadoEn',
          'creadoPor',
          'actualizadoEn',
          'actualizadoPor',
          'bajaEn',
          'bajaPor',
          'motivoBaja'
        ]) &&
        request.resource.data.email == docenteId &&
        request.resource.data.email is string &&
        request.resource.data.email.size() >= 5 &&
        request.resource.data.email.size() <= 254 &&
        request.resource.data.nombre is string &&
        request.resource.data.nombre.size() >= 1 &&
        request.resource.data.nombre.size() <= 120 &&
        request.resource.data.rol is string &&
        request.resource.data.activo is bool &&
        request.resource.data.creadoEn is timestamp &&
        request.resource.data.creadoPor is string &&
        request.resource.data.actualizadoEn == request.time &&
        request.resource.data.actualizadoPor == authenticatedEmail() &&
        (
          request.resource.data.bajaEn == null ||
          request.resource.data.bajaEn is timestamp
        ) &&
        request.resource.data.bajaPor is string &&
        request.resource.data.motivoBaja is string &&
        request.resource.data.motivoBaja.size() <= 300 &&
        (
          docenteId == 'ipem146centenario@gmail.com'
            ? (
              request.resource.data.rol == 'administrador' &&
              request.resource.data.activo == true
            )
            : request.resource.data.rol == 'docente'
        );
    }


    // =========================================================
    // CAMPOS MODIFICABLES DEL ALUMNO
    // =========================================================

    function studentMutableFields() {
      return [
        'estudiante',
        'salidasPestana',
        'eventosSalidasPestana',
        'pantallaBloqueada',
        'pantallaBloqueadaEn',
        'pantallaBloqueadaSalidas',
        'pantallaBloqueadaSeccion',
        'codigos',
        'tiemposRestantes',
        'finalizadas',
        'historialResultados',
        'chatIA',
        'ayudasComprension',
        'contadorPrevisualizaciones',
        'seccionActiva',
        'versionApp',
        'nombreGoogle',
        'fotoGoogle',
        'emailVerificado',
        'seguimientoExtension',
        'jitsiParticipacion',
        'actualizadoEn'
      ];
    }

    function newStudentAllowedFields() {
      return [
        'estudiante',
        'salidasPestana',
        'eventosSalidasPestana',
        'codigos',
        'tiemposRestantes',
        'finalizadas',
        'historialResultados',
        'chatIA',
        'ayudasComprension',
        'contadorPrevisualizaciones',
        'seccionActiva',
        'versionApp',
        'nombreGoogle',
        'fotoGoogle',
        'emailVerificado',
        'seguimientoExtension',
        'actualizadoEn',
        'uid',
        'email',
        'estadoCuenta',
        'solicitudAprobacionEn'
      ];
    }


    // =========================================================
    // REGISTRO DE ALUMNO
    // =========================================================

    function validNewStudent(uid) {
      return ownsDocument(uid) &&
        request.resource.data.keys().hasAll([
          'uid',
          'email',
          'estudiante',
          'estadoCuenta'
        ]) &&
        request.resource.data.keys().hasOnly(newStudentAllowedFields()) &&
        request.resource.data.uid == uid &&
        request.resource.data.email ==
          request.auth.token.get('email', '') &&
        request.resource.data.get('emailVerificado', false) ==
          request.auth.token.get('email_verified', false) &&
        request.resource.data.estadoCuenta == 'pendiente';
    }


    // =========================================================
    // ACTUALIZACIÓN DEL ALUMNO
    // =========================================================

    function validStudentUpdate(uid) {
      return ownsDocument(uid) &&
        request.resource.data.uid == resource.data.uid &&
        request.resource.data.email == resource.data.email &&
        request.resource.data.get('emailVerificado', false) ==
          request.auth.token.get('email_verified', false) &&

        // El alumno puede activar el bloqueo,
        // pero nunca quitar uno existente.
        (
          resource.data.get('pantallaBloqueada', false) != true ||
          request.resource.data.get('pantallaBloqueada', false) == true
        ) &&

        (
          !request.resource.data.diff(resource.data)
            .affectedKeys()
            .hasAny(['seguimientoExtension']) ||

          (
            request.resource.data.seguimientoExtension is map &&

            request.resource.data.seguimientoExtension.keys().hasAll([
              'instalada',
              'activa',
              'version',
              'ultimaSenalEn',
              'dominioActual',
              'tituloActual',
              'visitaIniciadaEn',
              'salidaGrupoId',
              'pendientes',
              'claseId'
            ]) &&

            request.resource.data.seguimientoExtension.keys().hasOnly([
              'instalada',
              'activa',
              'version',
              'ultimaSenalEn',
              'dominioActual',
              'tituloActual',
              'visitaIniciadaEn',
              'salidaGrupoId',
              'pendientes',
              'claseId'
            ]) &&

            request.resource.data.seguimientoExtension.instalada is bool &&
            request.resource.data.seguimientoExtension.activa is bool &&
            request.resource.data.seguimientoExtension.version is string &&
            request.resource.data.seguimientoExtension.version.size() <= 30 &&
            request.resource.data.seguimientoExtension.ultimaSenalEn is string &&
            request.resource.data.seguimientoExtension.ultimaSenalEn.size() <= 40 &&
            request.resource.data.seguimientoExtension.dominioActual is string &&
            request.resource.data.seguimientoExtension.dominioActual.size() <= 120 &&
            request.resource.data.seguimientoExtension.tituloActual is string &&
            request.resource.data.seguimientoExtension.tituloActual.size() <= 300 &&
            request.resource.data.seguimientoExtension.visitaIniciadaEn is string &&
            request.resource.data.seguimientoExtension.visitaIniciadaEn.size() <= 40 &&
            request.resource.data.seguimientoExtension.salidaGrupoId is string &&
            request.resource.data.seguimientoExtension.salidaGrupoId.size() <= 160 &&
            request.resource.data.seguimientoExtension.pendientes is number &&
            request.resource.data.seguimientoExtension.pendientes >= 0 &&
            request.resource.data.seguimientoExtension.pendientes <= 500 &&
            request.resource.data.seguimientoExtension.claseId is string &&
            request.resource.data.seguimientoExtension.claseId.size() <= 120
          )
        ) &&

        (
          !request.resource.data.diff(resource.data)
            .affectedKeys()
            .hasAny(['jitsiParticipacion']) ||

          (
            request.resource.data.jitsiParticipacion is map &&

            request.resource.data.jitsiParticipacion.keys().hasAll([
              'llamadaId',
              'tipo',
              'estado',
              'actualizadoEn'
            ]) &&

            request.resource.data.jitsiParticipacion.keys().hasOnly([
              'llamadaId',
              'tipo',
              'estado',
              'actualizadoEn'
            ]) &&

            request.resource.data.jitsiParticipacion.llamadaId is string &&
            request.resource.data.jitsiParticipacion.llamadaId.size() > 0 &&
            request.resource.data.jitsiParticipacion.llamadaId.size() <= 180 &&

            request.resource.data.jitsiParticipacion.tipo in [
              'individual',
              'grupal'
            ] &&

            request.resource.data.jitsiParticipacion.estado in [
              'notificado',
              'unido',
              'rechazado',
              'salio',
              'finalizada'
            ] &&

            request.resource.data.jitsiParticipacion.actualizadoEn ==
              request.time &&

            (
              (
                request.resource.data.jitsiParticipacion.tipo ==
                  'individual' &&
                resource.data.get('jitsiSala', {}).get('id', '') ==
                  request.resource.data.jitsiParticipacion.llamadaId
              ) ||

              (
                request.resource.data.jitsiParticipacion.tipo ==
                  'grupal' &&
                resource.data.get('jitsiGrupal', {}).get('id', '') ==
                  request.resource.data.jitsiParticipacion.llamadaId
              )
            )
          )
        ) &&

        request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(studentMutableFields());
    }


    function validIncompleteStudentRegistration(uid) {
      return ownsDocument(uid) &&
        (
          resource.data.get('uid', '') == '' ||
          resource.data.get('email', '') == '' ||
          !resource.data.keys().hasAny(['estadoCuenta'])
        ) &&
        validNewStudent(uid);
    }


    function validStudentResubmission(uid) {
      return ownsDocument(uid) &&
        resource.data.estadoCuenta == 'rechazado' &&
        request.resource.data.estadoCuenta == 'pendiente' &&
        request.resource.data.uid == resource.data.uid &&
        request.resource.data.email == resource.data.email &&
        request.resource.data.emailVerificado ==
          request.auth.token.get('email_verified', false) &&

        request.resource.data.keys().hasAll([
          'estudiante',
          'solicitudAprobacionEn',
          'antecedentesRechazo'
        ]) &&

        request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly([
            'estudiante',
            'emailVerificado',
            'estadoCuenta',
            'solicitudAprobacionEn',
            'antecedentesRechazo',
            'actualizadoEn'
          ]);
    }


    function validCompatibleMessageStatusUpdate(uid) {
      return ownsDocument(uid) &&
        resource.data.get('mensajeDocenteActual', {}) is map &&
        request.resource.data.get('mensajeDocenteActual', {}) is map &&
        request.resource.data.diff(resource.data)
          .affectedKeys()
          .hasOnly(['mensajeDocenteActual']) &&

        request.resource.data.mensajeDocenteActual
          .diff(resource.data.mensajeDocenteActual)
          .affectedKeys()
          .hasOnly([
            'recibido',
            'recibidoEn',
            'leido',
            'leidoEn'
          ]) &&

        (
          !request.resource.data.mensajeDocenteActual
            .diff(resource.data.mensajeDocenteActual)
            .affectedKeys()
            .hasAny(['recibido', 'recibidoEn']) ||

          (
            request.resource.data.mensajeDocenteActual.recibido == true &&
            request.resource.data.mensajeDocenteActual.recibidoEn ==
              request.time
          )
        ) &&

        (
          !request.resource.data.mensajeDocenteActual
            .diff(resource.data.mensajeDocenteActual)
            .affectedKeys()
            .hasAny(['leido', 'leidoEn']) ||

          (
            request.resource.data.mensajeDocenteActual.leido == true &&
            request.resource.data.mensajeDocenteActual.leidoEn ==
              request.time &&
            request.resource.data.mensajeDocenteActual.recibido == true
          )
        );
    }


    // =========================================================
    // ESTUDIANTES
    // =========================================================

    match /estudiantes/{uid} {

      allow get: if ownsDocument(uid) || isTeacher();

      allow list: if isTeacher();

      allow create: if validNewStudent(uid) || isTeacher();

      allow update: if
        validStudentUpdate(uid) ||
        validIncompleteStudentRegistration(uid) ||
        validStudentResubmission(uid) ||
        validCompatibleMessageStatusUpdate(uid) ||
        isTeacher();

      allow delete: if isTeacher();
    }


    // =========================================================
    // HISTORIAL DE DESBLOQUEOS
    // =========================================================

    match /estudiantes/{uid}/historialDesbloqueos/{eventoId} {

      allow get, list: if isTeacher();

      allow create: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'id',
          'estudianteUid',
          'desbloqueadoPor',
          'desbloqueadoPorUid',
          'desbloqueadoEn',
          'motivo'
        ]) &&
        request.resource.data.id == eventoId &&
        request.resource.data.estudianteUid == uid &&
        request.resource.data.desbloqueadoPorUid == request.auth.uid &&
        request.resource.data.motivo is string &&
        request.resource.data.motivo.size() > 0 &&
        request.resource.data.motivo.size() <= 500;

      allow update: if false;

      allow delete: if isTeacher();
    }


    // =========================================================
    // HISTORIAL DE DESCUENTOS
    // =========================================================

    match /estudiantes/{uid}/historialDescuentos/{eventoId} {

      allow get, list: if isTeacher();

      allow create: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'id',
          'estudianteUid',
          'docente',
          'docenteUid',
          'cambiadoEn',
          'valorAnterior',
          'valorNuevo',
          'motivoAnterior',
          'motivoNuevo'
        ]) &&

        request.resource.data.id == eventoId &&
        request.resource.data.estudianteUid == uid &&
        request.resource.data.docenteUid == request.auth.uid &&
        request.resource.data.docente ==
          request.auth.token.get('email', '') &&
        request.resource.data.cambiadoEn == request.time &&

        request.resource.data.valorAnterior is number &&
        request.resource.data.valorAnterior >= 0 &&
        request.resource.data.valorAnterior <= 10 &&

        request.resource.data.valorNuevo is number &&
        request.resource.data.valorNuevo >= 0 &&
        request.resource.data.valorNuevo <= 10 &&

        request.resource.data.motivoAnterior is string &&
        request.resource.data.motivoAnterior.size() <= 500 &&

        request.resource.data.motivoNuevo is string &&
        request.resource.data.motivoNuevo.size() <= 500;

      allow update: if false;

      allow delete: if isTeacher();
    }


    // =========================================================
    // HISTORIAL DE NOTAS
    // =========================================================

    match /estudiantes/{uid}/historialNotasDesafios/{eventoId} {

      allow get, list: if ownsDocument(uid) || isTeacher();

      allow create: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'id',
          'estudianteUid',
          'sectionId',
          'tipo',
          'valorAnterior',
          'valorNuevo',
          'notaAutomatica',
          'motivo',
          'docente',
          'docenteUid',
          'cambiadoEn'
        ]) &&

        request.resource.data.id == eventoId &&
        request.resource.data.estudianteUid == uid &&
        request.resource.data.sectionId is string &&
        request.resource.data.sectionId.size() > 0 &&
        request.resource.data.sectionId.size() <= 120 &&

        request.resource.data.tipo in [
          'modificacion',
          'restauracion'
        ] &&

        (
          request.resource.data.valorAnterior == null ||
          (
            request.resource.data.valorAnterior is number &&
            request.resource.data.valorAnterior >= 0 &&
            request.resource.data.valorAnterior <= 10
          )
        ) &&

        (
          request.resource.data.valorNuevo == null ||
          (
            request.resource.data.valorNuevo is number &&
            request.resource.data.valorNuevo >= 0 &&
            request.resource.data.valorNuevo <= 10
          )
        ) &&

        (
          request.resource.data.notaAutomatica == null ||
          (
            request.resource.data.notaAutomatica is number &&
            request.resource.data.notaAutomatica >= 0 &&
            request.resource.data.notaAutomatica <= 10
          )
        ) &&

        request.resource.data.motivo is string &&
        request.resource.data.motivo.size() <= 1000 &&

        request.resource.data.docente ==
          request.auth.token.get('email', '') &&

        request.resource.data.docenteUid == request.auth.uid &&
        request.resource.data.cambiadoEn == request.time;

      allow update: if false;

      allow delete: if isTeacher();
    }


    // =========================================================
    // HISTORIAL DE PESTAÑAS
    // =========================================================

    match /estudiantes/{uid}/historialPestanas/{eventoId} {

      allow get, list: if ownsDocument(uid) || isTeacher();

      allow create: if ownsDocument(uid) &&
        exists(/databases/$(database)/documents/estudiantes/$(uid)) &&
        get(/databases/$(database)/documents/estudiantes/$(uid))
          .data.get('estadoCuenta', 'activo') == 'activo' &&

        request.resource.data.keys().hasAll([
          'id',
          'estudianteUid',
          'estudianteEmail',
          'claseId',
          'salidaGrupoId',
          'tituloOrigen',
          'seccionOrigen',
          'seccionTitulo',
          'tituloDestino',
          'dominioDestino',
          'salidaEn',
          'regresoEn',
          'duracionSegundos',
          'extensionVersion',
          'expiraEn',
          'registradoEn'
        ]) &&

        request.resource.data.id == eventoId &&
        request.resource.data.estudianteUid == uid &&

        request.resource.data.estudianteEmail ==
          request.auth.token.get('email', '') &&

        request.resource.data.registradoEn == request.time &&

        request.resource.data.claseId is string &&
        request.resource.data.claseId.size() > 0 &&
        request.resource.data.claseId.size() <= 120 &&

        request.resource.data.salidaGrupoId is string &&
        request.resource.data.salidaGrupoId.size() > 0 &&
        request.resource.data.salidaGrupoId.size() <= 160 &&

        exists(
          /databases/$(database)/documents/controlClase/$(request.resource.data.claseId)
        ) &&

        get(
          /databases/$(database)/documents/controlClase/$(request.resource.data.claseId)
        ).data.iniciada == true &&

        request.resource.data.tituloOrigen is string &&
        request.resource.data.tituloOrigen.size() <= 300 &&

        request.resource.data.seccionOrigen is string &&
        request.resource.data.seccionOrigen.size() <= 120 &&

        request.resource.data.seccionTitulo is string &&
        request.resource.data.seccionTitulo.size() <= 200 &&

        request.resource.data.tituloDestino is string &&
        request.resource.data.tituloDestino.size() > 0 &&
        request.resource.data.tituloDestino.size() <= 300 &&

        request.resource.data.dominioDestino is string &&
        request.resource.data.dominioDestino.size() > 0 &&
        request.resource.data.dominioDestino.size() <= 120 &&

        request.resource.data.salidaEn is string &&
        request.resource.data.salidaEn.size() <= 40 &&

        request.resource.data.regresoEn is string &&
        request.resource.data.regresoEn.size() <= 40 &&

        request.resource.data.duracionSegundos is number &&
        request.resource.data.duracionSegundos >= 0 &&
        request.resource.data.duracionSegundos <= 86400 &&

        request.resource.data.extensionVersion is string &&
        request.resource.data.extensionVersion.size() <= 30 &&

        request.resource.data.expiraEn is timestamp &&
        request.resource.data.expiraEn > request.time;

      allow update: if false;

      allow delete: if isTeacher();
    }


    // =========================================================
    // REVISIONES DE PESTAÑAS
    // =========================================================

    match /estudiantes/{uid}/revisionesPestanas/{revisionId} {

      allow get, list: if isTeacher();

      allow create, update: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'salidaGrupoId',
          'estado',
          'motivo',
          'observacion',
          'etiquetas',
          'contabiliza',
          'docente',
          'docenteUid',
          'revisadoEn'
        ]) &&

        request.resource.data.keys().hasOnly([
          'salidaGrupoId',
          'estado',
          'motivo',
          'observacion',
          'etiquetas',
          'contabiliza',
          'docente',
          'docenteUid',
          'revisadoEn'
        ]) &&

        request.resource.data.salidaGrupoId is string &&
        request.resource.data.salidaGrupoId.size() > 0 &&
        request.resource.data.salidaGrupoId.size() <= 160 &&

        request.resource.data.estado in [
          'pendiente',
          'revisada',
          'justificada',
          'no_justificada',
          'permitida',
          'advertencia'
        ] &&

        request.resource.data.motivo is string &&
        request.resource.data.motivo.size() <= 120 &&

        request.resource.data.observacion is string &&
        request.resource.data.observacion.size() <= 1000 &&

        request.resource.data.etiquetas is list &&
        request.resource.data.etiquetas.size() <= 10 &&

        request.resource.data.contabiliza is bool &&

        request.resource.data.docente ==
          request.auth.token.get('email', '') &&

        request.resource.data.docenteUid == request.auth.uid &&
        request.resource.data.revisadoEn == request.time;

      allow delete: if isTeacher();
    }


    // =========================================================
    // REVISIONES VISIBLES AL ALUMNO
    // =========================================================

    match /estudiantes/{uid}/revisionesPestanasEstudiante/{revisionId} {

      allow get, list: if ownsDocument(uid) || isTeacher();

      allow create, update: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'salidaGrupoId',
          'estado',
          'revisado',
          'contabiliza',
          'revisadoEn',
          'actualizadoEn'
        ]) &&

        request.resource.data.keys().hasOnly([
          'salidaGrupoId',
          'estado',
          'revisado',
          'contabiliza',
          'revisadoEn',
          'actualizadoEn'
        ]) &&

        request.resource.data.salidaGrupoId is string &&
        request.resource.data.salidaGrupoId.size() > 0 &&
        request.resource.data.salidaGrupoId.size() <= 160 &&

        request.resource.data.estado in [
          'pendiente',
          'revisada',
          'justificada',
          'no_justificada',
          'permitida',
          'advertencia'
        ] &&

        request.resource.data.revisado is bool &&
        request.resource.data.contabiliza is bool &&
        request.resource.data.revisadoEn is timestamp &&
        request.resource.data.actualizadoEn == request.time;

      allow delete: if isTeacher();
    }


    // =========================================================
    // MENSAJES DOCENTE → ALUMNO
    // =========================================================

    match /estudiantes/{uid}/mensajesDocente/{mensajeId} {

      allow get, list: if ownsDocument(uid) || isTeacher();

      allow create: if isTeacher() &&
        request.resource.data.keys().hasAll([
          'id',
          'estudianteUid',
          'tipoDestino',
          'destinoDescripcion',
          'asunto',
          'contenido',
          'prioridad',
          'docente',
          'docenteUid',
          'enviadoEn',
          'recibido',
          'recibidoEn',
          'leido',
          'leidoEn'
        ]) &&

        request.resource.data.keys().hasOnly([
          'id',
          'estudianteUid',
          'tipoDestino',
          'destinoDescripcion',
          'asunto',
          'contenido',
          'prioridad',
          'docente',
          'docenteUid',
          'enviadoEn',
          'recibido',
          'recibidoEn',
          'leido',
          'leidoEn'
        ]) &&

        request.resource.data.id == mensajeId &&
